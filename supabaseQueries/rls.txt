ALTER TABLE "public"."Chat" ENABLE ROW LEVEL SECURITY;
ALTER TABLE "public"."Friendship" ENABLE ROW LEVEL SECURITY;
ALTER TABLE "public"."Message" ENABLE ROW LEVEL SECURITY;
ALTER TABLE "public"."OTP" ENABLE ROW LEVEL SECURITY;
ALTER TABLE "public"."Presence" ENABLE ROW LEVEL SECURITY;
ALTER TABLE "public"."Room" ENABLE ROW LEVEL SECURITY;
ALTER TABLE "public"."RoomMember" ENABLE ROW LEVEL SECURITY;
ALTER TABLE "public"."Signal" ENABLE ROW LEVEL SECURITY;
ALTER TABLE "public"."User" ENABLE ROW LEVEL SECURITY;


-- Chat table: Only allow users to access their own messages
CREATE POLICY "chat_select_policy"
  ON public."Chat"
  FOR SELECT
  USING ("sender_id" = auth.uid()::text OR "receiver_id" = auth.uid()::text);

-- Friendship table: Users can only see their own friendships
CREATE POLICY "friendship_select_policy"
  ON public."Friendship"
  FOR SELECT
  USING (
    "requesterId" = auth.uid()::text 
    OR "addresseeId" = auth.uid()::text
  );

-- Message table: Only allow users to access messages from rooms they belong to
CREATE POLICY "message_select_policy"
  ON public."Message"
  FOR SELECT
  USING ("user_id" = auth.uid()::text 
         OR "room_id" IN (SELECT "room_id" FROM public."RoomMember" WHERE "user_id" = auth.uid()::text));

-- OTP table: Allow users to view their own OTP records
CREATE POLICY "otp_select_policy"
  ON public."OTP"
  FOR SELECT
  USING ("username" = auth.uid()::text);

-- Presence table: Only show presence info for rooms the user belongs to
CREATE POLICY "presence_select_policy"
  ON public."Presence"
  FOR SELECT
  USING ("user_id" = auth.uid()::text 
         OR "room_id" IN (SELECT "room_id" FROM public."RoomMember" WHERE "user_id" = auth.uid()::text));

-- Room table: Users can access rooms they created or rooms they are members of
CREATE POLICY "room_select_policy"
  ON public."Room"
  FOR SELECT
  USING ("created_by" = auth.uid()::text 
         OR "id" IN (SELECT "room_id" FROM public."RoomMember" WHERE "user_id" = auth.uid()::text));

-- RoomMember table: Users can only see their own memberships
CREATE POLICY "room_member_select_policy"
  ON public."RoomMember"
  FOR SELECT
  USING ("user_id" = auth.uid()::text);

-- Signal table: Users can only access signals sent or received by them
CREATE POLICY "signal_select_policy"
  ON public."Signal"
  FOR SELECT
  USING ("sender_id" = auth.uid()::text 
         OR "receiver_id" = auth.uid()::text);

-- User table: Allow users to see their own data
CREATE POLICY "user_select_policy"
  ON public."User"
  FOR SELECT
  USING ("id" = auth.uid()::text);
