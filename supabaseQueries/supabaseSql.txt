CREATE TABLE public.Chat (
  id bigint NOT NULL DEFAULT nextval('"Chat_id_seq"'::regclass),
  sender_id text NOT NULL,
  receiver_id text NOT NULL,
  content text NOT NULL,
  created_at timestamp without time zone NOT NULL DEFAULT CURRENT_TIMESTAMP,
  edited_at timestamp without time zone,
  is_read boolean NOT NULL DEFAULT false,
  CONSTRAINT Chat_pkey PRIMARY KEY (id),
  CONSTRAINT Chat_sender_id_fkey FOREIGN KEY (sender_id) REFERENCES public.User(id),
  CONSTRAINT Chat_receiver_id_fkey FOREIGN KEY (receiver_id) REFERENCES public.User(id)
);

  id integer NOT NULL DEFAULT nextval('"Friendship_id_seq"'::regclass),

  status USER-DEFINED NOT NULL DEFAULT 'pending'::"FriendshipStatus",
  CONSTRAINT Friendship_pkey PRIMARY KEY (id),
  CONSTRAINT Friendship_requesterId_fkey FOREIGN KEY (requesterId) REFERENCES public.User(id),
  CONSTRAINT Friendship_addresseeId_fkey FOREIGN KEY (addresseeId) REFERENCES public.User(id)
);
